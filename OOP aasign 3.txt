#include<iostream>
using namespace std;

class Employee
{
public:
    int employeeID;
    string name;
    string department;

    void showDetails()
    {
        cout << "------ Employee Information ------" << endl;
        cout << "Employee ID: " << employeeID << endl;
        cout << "Name: " << name << endl;
        cout << "Department: " << department << endl;
    }
};

int main()
{
    Employee emp1;

    emp1.employeeID = 245678;
    emp1.name = "Siddhi Pawar";
    emp1.department = "SOAI";

    emp1.showDetails();

    return 0;
}
